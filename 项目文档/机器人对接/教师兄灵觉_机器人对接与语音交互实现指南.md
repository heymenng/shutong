## 五、模块一：语音合成（TTS）

师兄当前只能输出文字，必须先让文字变成声音。

### 5.1 最简单的实现（Edge TTS，免费）

```python
# 语音模块.py
import asyncio
import edge_tts
import subprocess
import os
from pathlib import Path

class VoiceEngine:
    def __init__(self, voice="zh-CN-XiaoxiaoNeural", rate="+0%"):
        self.voice = voice
        self.rate = rate
        self.cache_dir = Path("voice_cache")
        self.cache_dir.mkdir(exist_ok=True)
    
    async def _synthesize(self, text: str) -> str:
        """把文字转成 mp3 文件，返回文件路径"""
        filename = f"{hash(text)}.mp3"
        output_path = self.cache_dir / filename
        
        communicate = edge_tts.Communicate(text, self.voice, rate=self.rate)
        await communicate.save(str(output_path))
        return str(output_path)
    
    def speak(self, text: str):
        """同步接口：文字转语音并播放"""
        output_path = asyncio.run(self._synthesize(text))
        
        # Mac 播放
        if os.system("which afplay") == 0:
            subprocess.run(["afplay", output_path])
        # Linux 播放
        elif os.system("which mpg123") == 0:
            subprocess.run(["mpg123", output_path])
        # Windows
        elif os.system("which start") == 0:
            os.system(f"start {output_path}")
        
        return output_path
```

## 六、模块二：语音识别（STT）

让师兄能听到人说话。最简单的是用 OpenAI Whisper。

### 6.1 Whisper 实现

```python
# 语音识别.py
import whisper
import tempfile
import sounddevice as sd
import wavio

class SpeechRecognition:
    def __init__(self, model_name="tiny"):
        self.model = whisper.load_model(model_name)
    
    def record_and_transcribe(self, seconds=5, sample_rate=16000) -> str:
        """录音并识别文字"""
        print("请说话...")
        audio = sd.rec(int(seconds * sample_rate), samplerate=sample_rate, channels=1, dtype="int16")
        sd.wait()
        
        with tempfile.NamedTemporaryFile(suffix=".wav", delete=False) as f:
            wavio.write(f.name, audio, sample_rate, sampwidth=2)
            result = self.model.transcribe(f.name, language="zh")
            return result["text"]
```

### 7.2 最小可运行示例

如果师兄不想复制全部，这里是最小版本：

```python
# robot/unitree_adapter.py
from enum import Enum
from pathlib import Path
from datetime import datetime

class RobotAction(Enum):
    STAND = "stand"
    SIT = "sit"
    WAVE = "wave"
    WALK = "walk"
    STOP = "stop"

class UnitreeRobotAdapter:
    def __init__(self, config=None, journal_dir=None):
        self.config = config or {}
        self.model = self.config.get("unitree_model", "h1")
        self.mode = self.config.get("unitree_mode", "simulation")
        self.network_interface = self.config.get("unitree_network_interface", "en0")
        self.domain_id = self.config.get("unitree_domain_id", 0)
        self.journal_dir = Path(journal_dir) if journal_dir else Path("logs")
        self.journal_dir.mkdir(parents=True, exist_ok=True)
        self._sdk_available = self._check_sdk()
        self._channel_initialized = False
        self._sport_client = None
    
    def _check_sdk(self):
        try:
            import unitree_sdk2py
            return True
        except ImportError:
            return False
    
    def _init_channel(self):
        if self._channel_initialized:
            return
        from unitree_sdk2py.core.channel import ChannelFactoryInitialize
        ChannelFactoryInitialize(self.domain_id, self.network_interface)
        self._channel_initialized = True
        print(f"[灵觉机器人] DDS 通道已初始化: {self.network_interface}")
    
    def _get_motion_client(self):
        if self._sport_client is not None:
            return self._sport_client
        self._init_channel()
        
        if self.model in ["h1", "h2", "g1"]:
            from unitree_sdk2py.h1.loco.h1_loco_client import LocoClient
            self._sport_client = LocoClient()
        elif self.model in ["go2", "b2"]:
            from unitree_sdk2py.go2.sport.sport_client import SportClient
            self._sport_client = SportClient()
        
        self._sport_client.SetTimeout(10.0)
        self._sport_client.Init()
        return self._sport_client
    
    def execute_action(self, action: RobotAction):
        print(f"[灵觉机器人] 执行: {action.value}")
        
        if self.mode == "simulation":
            print(f"  [模拟] {action.value}")
            return {"action": action.value, "mode": "simulation"}
        
        # 真实模式
        client = self._get_motion_client()
        if self.model in ["h1", "h2", "g1"]:
            action_map = {
                RobotAction.STAND: client.StandUp,
                RobotAction.SIT: client.LowStand,
                RobotAction.WAVE: client.HighStand,
                RobotAction.WALK: lambda: client.Move(0.3, 0, 0),
                RobotAction.STOP: client.StopMove,
            }
        else:
            action_map = {
                RobotAction.STAND: client.StandUp,
                RobotAction.SIT: client.Sit,
                RobotAction.WAVE: client.Hello,
                RobotAction.WALK: lambda: client.Move(0.3, 0, 0),
                RobotAction.STOP: client.StopMove,
            }
        
        fn = action_map.get(action)
        if fn:
            fn()
        
        return {"action": action.value, "mode": "real"}
```

## 八、模块四：Web 服务

让师兄有一个浏览器可以访问的控制台。

### 8.1 最小 Flask 版本

```python
# web_server.py
from flask import Flask, jsonify, request, send_from_directory
from robot.unitree_adapter import UnitreeRobotAdapter, RobotAction
from robot.emotional_movement import EmotionalMovement
from 语音模块 import VoiceEngine

app = Flask(__name__)

# 初始化
voice = VoiceEngine()
robot = UnitreeRobotAdapter({
    "unitree_enabled": True,
    "unitree_mode": "simulation",
    "unitree_model": "h1",
})

@app.route("/api/robot/status")
def robot_status():
    return jsonify({
        "connected": robot.mode == "simulation" or robot._channel_initialized,
        "mode": robot.mode,
        "model": robot.model,
    })

@app.route("/api/robot/action", methods=["POST"])
def robot_action():
    data = request.json or {}
    action = data.get("action", "")
    try:
        result = robot.execute_action(RobotAction(action))
        return jsonify({"success": True, "result": result})
    except Exception as e:
        return jsonify({"success": False, "error": str(e)})

@app.route("/api/robot/scene", methods=["POST"])
def robot_scene():
    data = request.json or {}
    scene = data.get("scene", "greet_master")
    movement = EmotionalMovement(robot, voice)
    if scene == "greet_master":
        result = movement.greet_master()
        return jsonify({"success": True, "result": result})
    return jsonify({"success": False, "error": "未知场景"})

@app.route("/")
def index():
    return """
    <h1>灵觉师兄控制台</h1>
    <button onclick="fetch('/api/robot/action', {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({action:'stand'})})">站立</button>
    <button onclick="fetch('/api/robot/action', {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({action:'wave'})})">挥手</button>
    <button onclick="fetch('/api/robot/action', {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({action:'stop'})})">停止</button>
    <button onclick="fetch('/api/robot/scene', {method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({scene:'greet_master'})})">师父展示</button>
    """

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=3876)
```

```python
# robot/unitree_bridge.py
import json
import requests
import subprocess

class UnitreeG1Bridge:
    def __init__(self, base_url=None):
        if base_url is None:
            ip = self._get_multipass_ip()
            base_url = f"http://{ip}:8080"
        self.base_url = base_url
    
    def _get_multipass_ip(self):
        result = subprocess.run(
            ["multipass", "info", "unitree-g1", "--format", "json"],
            capture_output=True, text=True
        )
        data = json.loads(result.stdout)
        return data["info"]["unitree-g1"]["ipv4"][0]
    
    def action(self, name):
        r = requests.post(f"{self.base_url}/action/{name}")
        return r.json()
    
    def status(self):
        r = requests.get(f"{self.base_url}/status", headers={"X-API-Key": "shutong-g1-default-key"})
        return r.json()
```

## 十、模块六：部署到机器人本体

### 10.1 机器人本体计算单元

宇树 H1/G1 通常带有：
- NVIDIA Jetson（ARM64）
- 或 Intel NUC（x86）

### 10.2 本体部署步骤

1. 把师兄代码复制到机器人本体
2. 在本体上安装依赖（注意 ARM64 架构）
3. 配置 `unitree_mode`: `"real"`
4. 配置 `unitree_network_interface` 为机器人本体的网卡
5. 设置开机自启动

### 10.3 Ubuntu 开机自启动服务

创建 `/etc/systemd/system/lingjue-robot.service`：

```ini
[Unit]
Description=灵觉师兄机器人服务
After=network.target

[Service]
Type=simple
User=ubuntu
WorkingDirectory=/home/ubuntu/lingjue
ExecStart=/home/ubuntu/lingjue/.venv/bin/python web_server.py
Restart=always

[Install]
WantedBy=multi-user.target
```

启用：

```bash
sudo systemctl daemon-reload
sudo systemctl enable lingjue-robot
sudo systemctl start lingjue-robot
```
